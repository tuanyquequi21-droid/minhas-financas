function renderizarAgenda() { const container = document.getElementById('calendarioAgenda'); if (!container) return; container.innerHTML = ''; const diasSemana = ['Dom', 'Seg', 'Ter', 'Qua', 'Qui', 'Sex', 'Sáb']; // Cria cabeçalho diasSemana.forEach(dia => { const div = document.createElement('div'); div.className = 'dia-header'; div.innerText = dia; container.appendChild(div); }); // Gera os dias do mês atual selecionado for (let i = 1; i <= 30; i++) { const cardDia = document.createElement('div'); cardDia.className = 'dia-card'; cardDia.innerHTML = `${i}`; container.appendChild(cardDia); } }
